CREATE OR REPLACE TABLE DEFAULTDATABASE.FOODMART.EMPLOYEE (
    EMPLOYEE_ID        INTEGER,
    EMPLOYEE_NAME      VARCHAR(100),
    DEPARTMENT         VARCHAR(100)
);

CREATE OR REPLACE TABLE DEFAULTDATABASE.FOODMART.EMPLOYEETEST (
    EMPLOYEE_ID        INTEGER,
    FIRST_NAME         VARCHAR(100),
    LAST_NAME          VARCHAR(100),
    DEPARTMENT         VARCHAR(100),
    SALARY             NUMBER(10,2),
    HIRE_DATE          DATE
);

CREATE OR REPLACE TABLE DEFAULTDATABASE.FOODMART.DEPARTMENTINFO (
    DEPARTMENT_NAME    VARCHAR(100),
    DEPT_LOCATION      VARCHAR(100)
);
